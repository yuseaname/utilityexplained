deploy.yml
name: Deploy to Hostinger

on:
  push:
    branches: [main]
  workflow_dispatch:

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: peaceiris/actions-hugo@v3
        with:
          hugo-version: '0.141.0'
          extended: true
      - run: hugo --minify --gc
      - name: Verify image references (fail deploy on missing assets)
        run: python3 scripts/check_image_paths.py .
      - name: Verify rendered output (fail deploy on leaked shortcodes)
        run: python3 scripts/check_rendered_output.py public
      - name: Setup SSH
        run: |
          mkdir -p ~/.ssh
          echo "${{ secrets.HOSTINGER_SSH_KEY }}" > ~/.ssh/deploy_key
          chmod 600 ~/.ssh/deploy_key
          ssh-keygen -y -f ~/.ssh/deploy_key > /dev/null
          ssh-keyscan -H -p 65002 "${{ secrets.HOSTINGER_SSH_HOST }}" >> ~/.ssh/known_hosts
      - name: Deploy via rsync
        run: |
          rsync -avz --delete \
            -e "ssh -i ~/.ssh/deploy_key -p 65002" \
            ./public/ \
            "${{ secrets.HOSTINGER_SSH_USER }}@${{ secrets.HOSTINGER_SSH_HOST }}:~/domains/utilityexplained.com/public_html/"
      - name: Notify IndexNow (Bing/DDG ecosystem) of sitemap URLs
        if: github.event_name == 'push'
        run: |
          KEY="f8c073ef959f844532b388dda52d0bcd"
          URLS=$(python3 -c "
          import re
          xml = open('public/sitemap.xml').read()
          print('\n'.join(re.findall(r'<loc>([^<]+)</loc>', xml)))")
          printf '%s\n' "$URLS" | jq -R -s -c 'split("\n")[:-1]' > /tmp/urls.json
          jq -n --arg host "utilityexplained.com" --arg key "$KEY" --rawfile urls /tmp/urls.json \
            '{host:$host, key:$key, urlList:($urls|fromjson)}' > /tmp/payload.json
          curl -s -X POST "https://api.indexnow.org/indexnow" \
            -H "Content-Type: application/json; charset=utf-8" \
            -d @/tmp/payload.json -o /dev/null -w "IndexNow response: HTTP %{http_code}\n"
