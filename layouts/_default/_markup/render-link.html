{{- /* Amazon affiliate link normalizer (render-link hook). Every markdown to amazon.com /dp/ or /gp/ gets: 1. the site tag appended if missing (untagged links earned under Associates "Other" — 100% of Q3 orders, zero per-site attribution; see PORTFOLIO-PM 2026-09-26 entry A); 2. rel="sponsored nofollow noopener" per Associates requirements (shortcodes already do this); 3. data-rybbit affiliate_click tracking, format=inline_md. Implementation note: urls.Parse returns an immutable struct on Hugo 0.141 (field assignment fails to parse), so the href is rebuilt with plain string concatenation. */ -}} {{- $u := urls.Parse .Destination -}} {{- $isAmz := and (eq $u.Host "www.amazon.com") (or (strings.HasPrefix $u.Path "/dp/") (strings.HasPrefix $u.Path "/gp/")) -}} {{- if $isAmz -}} {{- $sep := "?" -}} {{- if strings.Contains .Destination "?" -}}{{- $sep = "&" -}}{{- end -}} {{- $href := .Destination -}} {{- if not $u.Query.tag -}} {{- $href = printf "%s%stag=%s" .Destination $sep (site.Params.amazon_tag | default "utexplained-20") -}} {{- end -}} {{ .Text }} {{- else -}} {{ .Text }} {{- end -}}
